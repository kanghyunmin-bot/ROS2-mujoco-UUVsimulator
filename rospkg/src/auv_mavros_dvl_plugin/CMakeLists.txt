cmake_minimum_required(VERSION 3.8)
project(auv_mavros_dvl_plugin)
find_package(ament_cmake REQUIRED)
find_package(mavros REQUIRED)
find_package(pluginlib REQUIRED)
find_package(auv_dvl_a50_msg REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(diagnostic_updater REQUIRED)
find_package(libmavconn REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(auv_mavros_dvl_plugin SHARED src/vision_position_delta.cpp)
target_compile_features(auv_mavros_dvl_plugin PUBLIC cxx_std_17)
ament_target_dependencies(auv_mavros_dvl_plugin mavros pluginlib auv_dvl_a50_msg
  tf2_ros tf2_eigen diagnostic_updater libmavconn mavros_msgs rclcpp sensor_msgs)
pluginlib_export_plugin_description_file(mavros plugins.xml)
install(TARGETS auv_mavros_dvl_plugin LIBRARY DESTINATION lib)
install(FILES LICENSE.md UPSTREAM.md DESTINATION share/${PROJECT_NAME})
ament_package()
